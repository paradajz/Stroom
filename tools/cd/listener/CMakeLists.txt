cmake_minimum_required(VERSION 3.21)
project(stroom_cd_listener LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The CD listener requires Linux and PipeWire")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3>=0.3.49)
pkg_check_modules(DBUS REQUIRED IMPORTED_TARGET dbus-1)
find_package(Threads REQUIRED)

# Generate native headers from the shared wire contracts.
get_filename_component(STROOM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/contracts")
foreach(contract audio cd_stream)
  set(input "${STROOM_ROOT}/shared/contracts/${contract}.json")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${input}")
  file(READ "${input}" definition)
  string(JSON count LENGTH "${definition}")
  math(EXPR last "${count} - 1")
  set(header "/* Generated from shared/contracts; do not edit. */\n#pragma once\n")
  foreach(index RANGE ${last})
    string(JSON key MEMBER "${definition}" ${index})
    string(JSON value GET "${definition}" "${key}" value)
    string(JSON type TYPE "${definition}" "${key}" value)
    if(type STREQUAL "STRING")
      string(REPLACE "\\" "\\\\" value "${value}")
      string(REPLACE "\"" "\\\"" value "${value}")
      string(REPLACE "\n" "\\n" value "${value}")
      set(value "\"${value}\"")
    endif()
    string(APPEND header "#define ${key} ${value}\n")
  endforeach()
  file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/contracts/${contract}.h" CONTENT "${header}" @ONLY)
endforeach()

set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/config.json")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/config.json" listener_config)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/listener_config.hpp"
  CONTENT "/* Generated from listener/config.json; do not edit. */\n#pragma once\ninline constexpr char listener_config_template[] = R\"stroom(${listener_config})stroom\";\n" @ONLY)

add_library(cd-listener-core STATIC protocol.cpp network.cpp config.cpp)
target_include_directories(cd-listener-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(cd-listener-core PUBLIC PkgConfig::PIPEWIRE Threads::Threads)
target_compile_options(cd-listener-core PRIVATE -Wall -Wextra -Werror)
add_executable(stroom-cd-listener main.cpp playback.cpp realtime.cpp)
target_link_libraries(stroom-cd-listener PRIVATE cd-listener-core PkgConfig::DBUS)
target_compile_options(stroom-cd-listener PRIVATE -Wall -Wextra -Werror)
